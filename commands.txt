.\.venv\Scripts\activate
python main.py

# Open in your web browser:
http://127.0.0.1:8000


# ==========================================
# 2. STOP / KILL RUNNING LOCALHOST (PORT 8000)
# ==========================================

# Option A: In PowerShell (Kill process using port 8000)
Stop-Process -Id (Get-NetTCPConnection -LocalPort 8000).OwningProcess -Force

# Option B: In Command Prompt (taskkill by port 8000)
for /f "tokens=5" %a in ('netstat -aon ^| findstr :8000 ^| findstr LISTENING') do taskkill /f /pid %a

# Option C: Kill any running Python process
taskkill /f /im python.exe
# (or in PowerShell: Stop-Process -Name python -Force)

# Option D: Double-click stop.bat
.\stop.bat


# ==========================================
# 3. CLEAN UP ALL TRACES
# ==========================================

.\cleanup.bat
